# /etc/nginx/sites-available/attendstack

server {
    listen 80;
    server_name attendance.nextgenapplication.com;

    # Increase max body size for file uploads if needed
    client_max_body_size 10M;

    # ==========================================
    # 1. PROXY ALL API REQUESTS TO DJANGO
    # ==========================================
    location /api {
        proxy_pass http://127.0.0.1:8001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # ==========================================
    # 2. SERVE DJANGO'S ADMIN PANEL & STATIC FILES
    # ==========================================
    location /admin {
        proxy_pass http://127.0.0.1:8001;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # Serve collected static files directly for better performance
    location /static/ {
        alias /home/squarefit/AttendStack/backend/staticfiles/;
    }

    # Serve Django's media uploads directly
    location /media/ {
        alias /home/squarefit/AttendStack/backend/media/;
    }

    # ==========================================
    # 3. PROXY ALL OTHER REQUESTS TO NEXT.JS
    # ==========================================
    location / {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
